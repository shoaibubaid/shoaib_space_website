# 15. Dynamic Programming
### 1. [Fibonacci](https://leetcode.com/problems/fibonacci-number/description/)

Given `n`, return the `n`th Fibonacci number, defined by `F(0) = 0`, `F(1) = 1`, and `F(n) = F(n-1) + F(n-2)` for `n > 1`.

```cpp
class Solution {
public:
    int fib(int n) {
        if(n < 2) return n;

        int prev2 = 0, prev1 = 1, curr;

        for(int i = 2; i <= n; i++) {
            curr = prev1 + prev2;
            prev2 = prev1;
            prev1 = curr;
        }

        return curr;
    }
};
```

### 2. [Climbing Stairs](https://leetcode.com/problems/climbing-stairs/description/)

You are climbing a staircase of `n` steps. Each move you can climb either `1` or `2` steps. Return the number of distinct ways to reach the top.

**Intuition**: the number of ways to reach step `i` is the sum of the ways to reach `i-1` and `i-2` — same recurrence as Fibonacci.

```cpp
class Solution {
public:
    int climbStairs(int n) {
        if(n <= 2) return n;

        int prev2 = 1, prev1 = 2, curr;

        for(int i = 3; i <= n; i++) {
            curr = prev1 + prev2;
            prev2 = prev1;
            prev1 = curr;
        }

        return curr;
    }
};
```

### 3. [House Robber](https://leetcode.com/problems/house-robber/)

You are a professional robber planning to rob houses along a street. Each house has a certain amount of money stashed, the only constraint stopping you from robbing each of them is that adjacent houses have security systems connected and **it will automatically contact the police if two adjacent houses were broken into on the same night.**

Given an integer array `nums` representing the amount of money of each house, return the *maximum amount of money* you can rob tonight without alerting the police.

```cpp
class Solution {
public:
    int rob(vector<int>& nums) {
        int n = nums.size();

        if(n == 1)
            return nums[0];

        int two_before = 0;
        int one_before = nums[0];
        int current;
        for(int i = 1; i < n; i++){
            // Taking increases or leaving
            current = max(one_before, two_before+nums[i]);

            two_before = one_before;
            one_before = current;
        }
        return current;
    }
};
```

### 4. [0/1 Knapsack](https://www.geeksforgeeks.org/problems/0-1-knapsack-problem0945/1)

Given weights and values of `n` items and a knapsack of capacity `W`, find the maximum value achievable by picking a subset of items such that the total weight does not exceed `W`. Each item can either be taken as a whole or left behind (no fractions), and each item can be used at most once.

**Memoization (top-down)**:

```cpp
class Solution {
public:
    vector<vector<int>> memo;

    int solve(int idx, int capacity, vector<int>& wt, vector<int>& val) {
        if(idx == 0) {
            return (wt[0] <= capacity) ? val[0] : 0;
        }

        if(memo[idx][capacity] != -1) return memo[idx][capacity];

        int notTake = solve(idx - 1, capacity, wt, val);

        int take = 0;
        if(wt[idx] <= capacity) {
            take = val[idx] + solve(idx - 1, capacity - wt[idx], wt, val);
        }

        return memo[idx][capacity] = max(take, notTake);
    }

    int knapsack(int W, vector<int>& val, vector<int>& wt) {
        int n = wt.size();
        memo.assign(n, vector<int>(W + 1, -1));
        return solve(n - 1, W, wt, val);
    }
};
```

### 5. [0/1 Knapsack Tabulation](https://www.geeksforgeeks.org/problems/0-1-knapsack-problem0945/1)

Same problem as above, solved bottom-up with a 2D table `dp[i][cap]` = best value using the first `i` items within capacity `cap`.

```cpp
class Solution {
public:
    int knapsack(int W, vector<int>& val, vector<int>& wt) {
        int n = wt.size();
        vector<vector<int>> dp(n, vector<int>(W + 1, 0));

        for(int cap = wt[0]; cap <= W; cap++) {
            dp[0][cap] = val[0];
        }

        for(int i = 1; i < n; i++) {
            for(int cap = 0; cap <= W; cap++) {
                int notTake = dp[i - 1][cap];

                int take = 0;
                if(wt[i] <= cap) {
                    take = val[i] + dp[i - 1][cap - wt[i]];
                }

                dp[i][cap] = max(take, notTake);
            }
        }

        return dp[n - 1][W];
    }
};
```

### 6. [Subset sum](https://www.geeksforgeeks.org/problems/subset-sum-problem-1611555638/1)

Given an array of non-negative integers and a target sum, determine whether there exists a subset of the array whose elements add up exactly to that target.

```cpp
class Solution {
public:
    bool isSubsetSum(vector<int>& arr, int target) {
        int n = arr.size();
        vector<vector<bool>> dp(n + 1, vector<bool>(target + 1, false));

        for(int i = 0; i <= n; i++) dp[i][0] = true; // sum 0 always achievable

        for(int i = 1; i <= n; i++) {
            for(int sum = 1; sum <= target; sum++) {
                bool notTake = dp[i - 1][sum];

                bool take = false;
                if(arr[i - 1] <= sum) {
                    take = dp[i - 1][sum - arr[i - 1]];
                }

                dp[i][sum] = take || notTake;
            }
        }

        return dp[n][target];
    }
};
```

### 7. [Target Sum](https://www.geeksforgeeks.org/problems/target-sum-1626326450/1)

Given an array of non-negative integers and a target value, count the number of subsets whose elements sum exactly to the target.

**Intuition**: identical transition to subset sum, but accumulate counts instead of a boolean.

```cpp
class Solution {
public:
    int perfectSum(vector<int>& arr, int target) {
        int n = arr.size();
        vector<vector<int>> dp(n + 1, vector<int>(target + 1, 0));

        for(int i = 0; i <= n; i++) dp[i][0] = 1; // empty subset sums to 0

        for(int i = 1; i <= n; i++) {
            for(int sum = 0; sum <= target; sum++) {
                int notTake = dp[i - 1][sum];

                int take = 0;
                if(arr[i - 1] <= sum) {
                    take = dp[i - 1][sum - arr[i - 1]];
                }

                dp[i][sum] = take + notTake;
            }
        }

        return dp[n][target];
    }
};
```

### 8. [Longest Increasing subsequence](https://leetcode.com/problems/longest-increasing-subsequence/)

Given an integer array `nums`, return the length of the longest strictly increasing subsequence.

```cpp
class Solution {
public:
    int lengthOfLIS(vector<int>& nums) {
        int n = nums.size();
        vector<int> dp(n, 1); // dp[i] = LIS ending at i

        int best = 1;

        for(int i = 1; i < n; i++) {
            for(int j = 0; j < i; j++) {
                if(nums[j] < nums[i]) {
                    dp[i] = max(dp[i], dp[j] + 1);
                }
            }
            best = max(best, dp[i]);
        }

        return best;
    }
};
```

### 9. [Longest Common subsequence](https://leetcode.com/problems/longest-common-subsequence/description/)

Given two strings `text1` and `text2`, return the length of their **longest common subsequence**. If there is no common subsequence, return `0`.

A subsequence of a string is a new string generated from the original string with some characters (can be none) deleted without changing the relative order of the remaining characters.

For example, `"ace"` is a subsequence of `"abcde"`.
A common subsequence of two strings is a subsequence that is common to both strings.

```cpp
class Solution {
public:
    int longestCommonSubsequence(string text1, string text2) {
        int m = text1.size();
        int n = text2.size();

        vector<vector<int>> dp(m+1, vector<int>(n+1, 0));

        for(int i = 1; i < m+1; i++){
            for(int j = 1; j < n+1; j++){

                if(text1[i-1] == text2[j-1]){
                    dp[i][j] = 1 + dp[i-1][j-1];
                }
                else{
                    dp[i][j] = max(dp[i][j-1],dp[i-1][j]);
                }
            }
        }

        return dp[m][n];
    }
};
```

### 10. [Unique Paths](https://leetcode.com/problems/unique-paths/description/)

A robot sits at the top-left corner of an `m x n` grid. It can only move either down or right at each step, trying to reach the bottom-right corner. Return the number of unique paths.

```cpp
class Solution {
public:
    int uniquePaths(int m, int n) {
        vector<vector<int>> dp(m, vector<int>(n, 1));

        for(int i = 1; i < m; i++) {
            for(int j = 1; j < n; j++) {
                dp[i][j] = dp[i - 1][j] + dp[i][j - 1];
            }
        }

        return dp[m - 1][n - 1];
    }
};
```

### 11. [Buy Sell Stocks](https://leetcode.com/problems/best-time-to-buy-and-sell-stock/description/)

Given an array `prices` where `prices[i]` is the price of a stock on day `i`, you may complete **at most one** buy and one sell (buy before you sell). Return the maximum profit achievable, or `0` if no profit is possible.

```cpp
class Solution {
public:
    int maxProfit(vector<int>& prices) {
        int minPrice = INT_MAX;
        int best = 0;

        for(int price : prices) {
            minPrice = min(minPrice, price);
            best = max(best, price - minPrice);
        }

        return best;
    }
};
```

### 12. [Buy Sell Stocks 2](https://leetcode.com/problems/best-time-to-buy-and-sell-stock-ii/description/)

Same setup, but now you may complete **as many transactions as you like** (buy and sell multiple times), as long as you hold at most one share at a time. Return the maximum total profit.

**Intuition**: greedily capture every positive day-to-day increase.

```cpp
class Solution {
public:
    int maxProfit(vector<int>& prices) {
        int profit = 0;

        for(int i = 1; i < prices.size(); i++) {
            if(prices[i] > prices[i - 1]) {
                profit += prices[i] - prices[i - 1];
            }
        }

        return profit;
    }
};
```

### 13. [Buy Sell Stocks 3](https://leetcode.com/problems/best-time-to-buy-and-sell-stock-iii/description/)

Same setup, but you may complete **at most two transactions**. Return the maximum profit.

**Intuition**: track four running states — cost of first buy, profit after first sell, cost of second buy (net of first profit), and profit after second sell.

```cpp
class Solution {
public:
    int maxProfit(vector<int>& prices) {
        int buy1 = INT_MIN, sell1 = 0;
        int buy2 = INT_MIN, sell2 = 0;

        for(int price : prices) {
            buy1 = max(buy1, -price);
            sell1 = max(sell1, buy1 + price);
            buy2 = max(buy2, sell1 - price);
            sell2 = max(sell2, buy2 + price);
        }

        return sell2;
    }
};
```

### 14. [Buy Sell Stocks 4](https://leetcode.com/problems/best-time-to-buy-and-sell-stock-iv/description/)

Same setup, generalized to **at most `k` transactions**. Return the maximum profit.

```cpp
class Solution {
public:
    int maxProfit(int k, vector<int>& prices) {
        int n = prices.size();
        if(n == 0) return 0;

        vector<int> buy(k + 1, INT_MIN), sell(k + 1, 0);

        for(int price : prices) {
            for(int t = 1; t <= k; t++) {
                buy[t] = max(buy[t], sell[t - 1] - price);
                sell[t] = max(sell[t], buy[t] + price);
            }
        }

        return sell[k];
    }
};
```

### 15. [Min cost to cut stick](https://leetcode.com/problems/minimum-cost-to-cut-a-stick/)

<font color="red">**HARD**</font>

Given a wooden stick of length `n` and an array `cuts` listing the positions where cuts must be made (in any order you choose), the cost of one cut equals the length of the stick segment being cut through at that moment. Return the minimum total cost to perform all the cuts.

**Intuition**: add `0` and `n` as sentinel boundaries, sort the cut positions, then run an interval DP where `dp[i][j]` is the min cost to make all cuts strictly between boundary indices `i` and `j`; trying every cut as the "last one made" in that interval gives cost `(right - left) + dp[i][k] + dp[k][j]`.

```cpp
class Solution {
public:
    int minCost(int n, vector<int>& cuts) {
        cuts.push_back(0);
        cuts.push_back(n);
        sort(cuts.begin(), cuts.end());

        int m = cuts.size();
        vector<vector<int>> dp(m, vector<int>(m, 0));

        // len = distance between boundary indices
        for(int len = 2; len < m; len++) {
            for(int i = 0; i + len < m; i++) {
                int j = i + len;
                int best = INT_MAX;

                for(int k = i + 1; k < j; k++) {
                    int cost = (cuts[j] - cuts[i]) + dp[i][k] + dp[k][j];
                    best = min(best, cost);
                }

                dp[i][j] = best;
            }
        }

        return dp[0][m - 1];
    }
};
```
